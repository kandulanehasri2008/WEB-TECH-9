   Page sections and intro text use it, so we state it explicitly. */

   The element stays in the flow but can be nudged from its normal spot, and
   it becomes the anchor ("containing block") for absolutely positioned children. */

  toastText.textContent = message;

// ---------- Open / close panels (test tips and FAQ) ----------
var toggleButtons = document.querySelectorAll(".toggle-btn");

function setPanel(button, open) {
  var panel = document.getElementById(button.getAttribute("aria-controls"));
  if (open) {
    panel.classList.add("is-open");
    button.classList.add("is-open");
  } else {
    panel.classList.remove("is-open");
    button.classList.remove("is-open");
  }
  button.setAttribute("aria-expanded", open ? "true" : "false");
}

toggleButtons.forEach(function (button) {
  button.addEventListener("click", function () {
    var willOpen = !button.classList.contains("is-open");
    // FAQ works as an accordion: opening one closes the others
    var faq = button.closest(".faq");
    if (faq && willOpen) {
      faq.querySelectorAll(".toggle-btn").forEach(function (other) { setPanel(other, false); });
    }
    setPanel(button, willOpen);
  });
});

function buildSummary() {
  var data = new FormData(form);
  var tests = [];
  form.querySelectorAll('input[name="tests"]:checked').forEach(function (box) {
    tests.push(box.parentNode.textContent.trim());
  });
  var slot = form.elements["slot"];
  var parts = data.get("date").split("-");
  var niceDate = new Date(parts[0], parts[1] - 1, parts[2]).toLocaleDateString("en-IN", {
    day: "numeric", month: "long", year: "numeric"
  });

  summary.textContent = "";
  addSummaryRow("Name", data.get("name"));
  addSummaryRow("Date", niceDate);
  addSummaryRow("Time", slot.options[slot.selectedIndex].text);
  addSummaryRow("Package", packageSelect.value ? packageSelect.options[packageSelect.selectedIndex].text : "None");
  addSummaryRow("Tests", tests.length ? tests.join(", ") : "None selected");
  addSummaryRow("Collection", data.get("collection") === "home" ? "Home collection" : "Visit the centre");
}

var basketList = document.getElementById("basket-list");

function formatINR(amount) {
  return "INR " + amount.toLocaleString("en-IN");
}

// Number of people a test is booked for: a whole number from 1 to 10
function readQty(input) {
  var qty = parseInt(input.value, 10);
  if (isNaN(qty) || qty < 1) { return 1; }
  return Math.min(qty, 10);
}

// Build one basket row (li) from scratch
function createBasketItem(id, name, price, isCustom) {
  var li = document.createElement("li");
  li.className = "basket-item";
  li.setAttribute("data-id", id);
  li.setAttribute("data-name", name);
  li.setAttribute("data-price", price);
  li.setAttribute("data-custom", isCustom ? "true" : "false");

  var info = document.createElement("div");
  info.className = "item-info";
  var nameEl = document.createElement("span");
  nameEl.className = "item-name";
  nameEl.textContent = name;                       // textContent keeps typed text safe
  var priceEl = document.createElement("span");
  priceEl.className = "item-price";
  priceEl.textContent = isCustom ? "Price quoted at the centre" : formatINR(price) + " per person";
  info.appendChild(nameEl);
  info.appendChild(priceEl);

  var qtyLabel = document.createElement("label");
  qtyLabel.className = "qty";
  qtyLabel.appendChild(document.createTextNode("People"));
  var qtyInput = document.createElement("input");
  qtyInput.setAttribute("type", "number");
  qtyInput.setAttribute("min", "1");
  qtyInput.setAttribute("max", "10");
  qtyInput.setAttribute("value", "1");
  qtyInput.setAttribute("aria-label", "Number of people for " + name);
  qtyLabel.appendChild(qtyInput);

  var subtotal = document.createElement("span");
  subtotal.className = "item-subtotal";
  subtotal.textContent = isCustom ? "-" : formatINR(price);

  var removeBtn = document.createElement("button");
  removeBtn.setAttribute("type", "button");
  removeBtn.className = "remove-btn";
  removeBtn.textContent = "Remove";
  removeBtn.setAttribute("aria-label", "Remove " + name + " from basket");

  li.appendChild(info);
  li.appendChild(qtyLabel);
  li.appendChild(subtotal);
  li.appendChild(removeBtn);
  return li;
}

// Recalculate everything that is displayed from the current list items
function updateBasket() {
  var items = basketList.querySelectorAll(".basket-item");
  var total = 0;
  items.forEach(function (item) {
    if (item.dataset.custom === "true") { return; }
    var subtotal = Number(item.dataset.price) * readQty(item.querySelector(".qty input"));
    item.querySelector(".item-subtotal").textContent = formatINR(subtotal);
    total += subtotal;
  });

  var count = items.length;
  basketSummary.innerHTML = "<strong>" + count + "</strong> test" + (count === 1 ? "" : "s") + " in your basket";
  basketTotal.textContent = formatINR(total);
  basketCount.textContent = count;
  basketCount.hidden = count === 0;
  basketEmpty.hidden = count !== 0;
  basketSummary.hidden = count === 0;
  continueBtn.disabled = count === 0;
  clearBtn.disabled = count === 0;
}

function addToBasket(id, name, price, isCustom) {
  if (basketList.querySelector('[data-id="' + id + '"]')) { return false; }   // already there
  basketList.appendChild(createBasketItem(id, name, price, isCustom));
  updateBasket();
  return true;
}

function removeBasketItem(li) {
  if (li.classList.contains("removing")) { return; }
  var id = li.getAttribute("data-id");
  var name = li.getAttribute("data-name");
  li.classList.add("removing");                    // play the slide-out first
  setTimeout(function () {
    li.remove();
    var addBtn = document.querySelector('.add-test[data-id="' + id + '"]');
    if (addBtn) {
      addBtn.removeAttribute("disabled");
      addBtn.textContent = "Add";
    }
    updateBasket();
    showToast(name + " removed from your basket.");
  }, 250);
}

function clearBasket() {
  basketList.innerHTML = "";
  document.querySelectorAll(".add-test").forEach(function (btn) {
    btn.removeAttribute("disabled");
    btn.textContent = "Add";
  });
  updateBasket();
}

// "Add" buttons in the price list
document.querySelectorAll(".add-test").forEach(function (btn) {
  btn.addEventListener("click", function () {
    var added = addToBasket(btn.dataset.id, btn.dataset.name, Number(btn.dataset.price), false);
    if (added) {
      btn.setAttribute("disabled", "");
      btn.textContent = "Added";
      showToast(btn.dataset.name + " added to your basket.");
    }
  });
});

// One listener on the list handles every row: remove button and quantity changes
basketList.addEventListener("click", function (event) {
  var removeBtn = event.target.closest(".remove-btn");
  if (removeBtn) { removeBasketItem(removeBtn.closest(".basket-item")); }
});
basketList.addEventListener("input", updateBasket);
basketList.addEventListener("change", function (event) {
  if (event.target.matches(".qty input")) {
    event.target.value = readQty(event.target);    // tidy up empty or out-of-range values
    updateBasket();
  }
});

clearBtn.addEventListener("click", function () {
  clearBasket();
  showToast("Basket cleared.");
});

// Add a test that is not on the price list
function showCustomMessage(text, isError) {
  customMsg.textContent = text;
  customMsg.className = "hint " + (isError ? "error" : "ok");
  if (isError) {
    customInput.setAttribute("aria-invalid", "true");
  } else {
    customInput.removeAttribute("aria-invalid");
  }
}

function addCustomTest() {
  var name = customInput.value.trim().replace(/\s+/g, " ");
  if (name.length < 3) {
    showCustomMessage("Enter at least 3 characters for the test name.", true);
    customInput.focus();
    return;
  }
  var id = "custom-" + name.toLowerCase().replace(/[^a-z0-9]+/g, "-");
  if (!addToBasket(id, name, 0, true)) {
    showCustomMessage("That test is already in your basket.", true);
    return;
  }
  customInput.value = "";
  showCustomMessage(name + " added. We will quote the price at the centre.", false);
}

customAddBtn.addEventListener("click", addCustomTest);
customInput.addEventListener("keydown", function (event) {
  if (event.key === "Enter") { event.preventDefault(); addCustomTest(); }
});

// Copy the basket into the booking form
continueBtn.addEventListener("click", function () {
  form.querySelectorAll('input[name="tests"]').forEach(function (box) { box.checked = false; });

  var groupLines = [];
  var extras = [];
  basketList.querySelectorAll(".basket-item").forEach(function (item) {
    var qty = readQty(item.querySelector(".qty input"));
    if (item.dataset.custom === "true") {
      extras.push(item.dataset.name);
      return;
    }
    var box = form.querySelector('input[name="tests"][value="' + item.dataset.id + '"]');
    if (box) { box.checked = true; }
    if (qty > 1) { groupLines.push(item.dataset.name + " x" + qty); }
  });

  var lines = [];
  if (groupLines.length) { lines.push("Tests for several people: " + groupLines.join("; ") + "."); }
  if (extras.length) { lines.push("Additional tests requested: " + extras.join(", ") + "."); }

  var notesField = document.getElementById("notes");
  var base = notesField.value.split("[Basket]")[0].trim();      // keep what the patient typed
  var block = lines.length ? "[Basket] " + lines.join(" ") : "";
  notesField.value = ((base && block) ? base + "\n" : base) + block;
  notesField.value = notesField.value.slice(0, 300);

  showToast("Basket tests added to the booking form.");
  document.getElementById("book").scrollIntoView({ behavior: "smooth", block: "start" });
});
